% figstyle.tex — the single visual system for docs/analytic.
% Every figure in the document uses ONLY the colours, node styles and arrow
% styles defined here.  Do not define colours or styles inside a figure file.
% Semantic roles (see FIGURES.md §1):
%   machine  — Turing-machine side (tapes, states, δ, configurations, time)
%   term     — lambda side (terms, trees, redexes, closures, fuel)
%   desc     — descriptions as finite data (⟨M⟩, ser(t), quoted code, bytes)
%   check    — what the prototype EXECUTES / CHECKS
%   cited    — what remains a CITED analytic theorem (quantum soundness etc.)
%   focus    — the one thing the eye should land on (scanned cell, current redex)

\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,fit,backgrounds,matrix,calc,
  shapes.geometric,shapes.multipart,shapes.symbols,decorations.pathreplacing,
  decorations.markings,decorations.pathmorphing,chains,scopes,quotes,cd,trees,
  automata,patterns,patterns.meta,shadows.blur}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\usepackage[skins,breakable]{tcolorbox}
\usepackage[font=small,labelfont={sf,bf},labelsep=period,textfont={sf},
  justification=justified,singlelinecheck=false]{caption}
\usepackage{float}

% ---- palette (warm paper, warm ink; two same-lightness accents + violet + slate)
\definecolor{paper}{HTML}{FBF8F2}   % page-like fill for panels
\definecolor{ink}{HTML}{22201C}     % near-black, warm
\definecolor{inkmute}{HTML}{6B665E} % secondary text
\definecolor{rule}{HTML}{D9D3C7}    % hairlines, grids, tape borders
\definecolor{machine}{HTML}{1F6F78} % teal      — Turing side
\definecolor{machinelt}{HTML}{DCEEF0}
\definecolor{term}{HTML}{B5561D}    % rust      — lambda side
\definecolor{termlt}{HTML}{F7E4D6}
\definecolor{desc}{HTML}{5B4B8A}    % violet    — descriptions / quoted bytes
\definecolor{desclt}{HTML}{E6E1F1}
\definecolor{check}{HTML}{3D7A3A}   % green     — executed / checked
\definecolor{checklt}{HTML}{DFEEDC}
\definecolor{cited}{HTML}{8A857C}   % slate     — cited analytic theorem
\definecolor{citedlt}{HTML}{ECE9E2}
\definecolor{focus}{HTML}{F2C14E}   % amber     — the scanned cell / current redex
\definecolor{focuslt}{HTML}{FBEFC9}
\definecolor{bad}{HTML}{B03A2E}     % red       — error / TypeError / wrong path

  cell head/.style={cell, fill=focuslt, draw=focus, line width=0.9pt},

  state initial/.style={state, fill=white},

  step/.style={->, machine, line width=0.8pt},

  tleaf/.style={tnode, fill=white},
  redex/.style={tnode, fill=focuslt, draw=focus, line width=1pt},
  tedge/.style={draw=term, line width=0.7pt},
  bind/.style={->, term, dashed, line width=0.6pt, bend left=40, shorten >=1pt},

  descbox/.style={draw=desc, line width=0.8pt, fill=desclt, rounded corners=2pt,
    inner sep=5pt, align=center},
  bytes/.style={draw=desc, fill=white, minimum height=5mm, inner sep=1.5pt,

  xformlbl/.style={lbl, above, midway, text=ink, font=\footnotesize\sffamily\bfseries},
